# cppcheck suppressions for BookHub (C++23 / Qt6)
#
# Usage:
#   cppcheck --suppressions-list=cppcheck-suppressions.txt \
#            --enable=all --std=c++23 \
#            --suppress=missingIncludeSystem \
#            -I src/ src/
#
# Format: errorId[:filename[:lineNumber]]
#   errorId alone  — suppress globally
#   errorId:*glob* — suppress in matching files

# ---------------------------------------------------------------------------
# Qt header resolution
# ---------------------------------------------------------------------------
# Qt headers are not on cppcheck's default search path. These would flood the
# output with thousands of irrelevant warnings.
missingIncludeSystem
missingInclude

# ---------------------------------------------------------------------------
# Qt MOC-generated files
# ---------------------------------------------------------------------------
# Files like moc_main_window.cpp are auto-generated by AUTOMOC and contain
# C-style casts, unusual patterns, and functions that appear unused from
# cppcheck's view. None of these are real defects.
cstyleCast:*moc_*
unusedFunction:*moc_*
noExplicitConstructor:*moc_*
uninitMemberVar:*moc_*

# ---------------------------------------------------------------------------
# Test files
# ---------------------------------------------------------------------------
# Test helpers intentionally define functions that are only called by the
# test framework via QTest macros, making them look unused to cppcheck.
unusedFunction:tests/*

# ---------------------------------------------------------------------------
# Third-party / generated paths
# ---------------------------------------------------------------------------
# build/ is never part of the source tree but may appear if cppcheck is
# pointed at the wrong directory. Suppress everything there as a safety net.
*:*/build/*
